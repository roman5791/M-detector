cmake_minimum_required(VERSION 3.8)
project(m_detector)

# Default to C++17
if(NOT CMAKE_CXX_STANDARD)
  set(CMAKE_CXX_STANDARD 17)
endif()

if(CMAKE_COMPILER_IS_GNUCXX OR CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()

SET(CMAKE_BUILD_TYPE "Release")

add_definitions(-DROOT_DIR=\"${CMAKE_CURRENT_SOURCE_DIR}/\")

set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -fexceptions")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -std=c++17 -ltbb -pthread -fexceptions")

message("Current CPU architecture: ${CMAKE_SYSTEM_PROCESSOR}")
if(CMAKE_SYSTEM_PROCESSOR MATCHES "(x86)|(X86)|(amd64)|(AMD64)")
  include(ProcessorCount)
  ProcessorCount(N)
  message("Processor number: ${N}")
  if(N GREATER 5)
    add_definitions(-DMP_EN)
    add_definitions(-DMP_PROC_NUM=4)
    message("core for MP: 4")
  elseif(N GREATER 3)
    math(EXPR PROC_NUM "${N} - 2")
    add_definitions(-DMP_EN)
    add_definitions(-DMP_PROC_NUM="${PROC_NUM}")
    message("core for MP: ${PROC_NUM}")
  else()
    add_definitions(-DMP_PROC_NUM=1)
  endif()
else()
  add_definitions(-DMP_PROC_NUM=1)
endif()

find_package(OpenMP QUIET)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} ${OpenMP_CXX_FLAGS}")
set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} ${OpenMP_C_FLAGS}")

# Find ROS2 dependencies
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_eigen REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(pcl_ros REQUIRED)

# Find system dependencies
find_package(Eigen3 REQUIRED)
find_package(PCL 1.8 REQUIRED)
find_package(TBB REQUIRED)
find_package(OpenCV REQUIRED)
find_package(PythonLibs REQUIRED)

message(Eigen: ${EIGEN3_INCLUDE_DIR})

include_directories(
  ${EIGEN3_INCLUDE_DIR}
  ${PCL_INCLUDE_DIRS}
  ${PYTHON_INCLUDE_DIRS}
  include
)

# ROS2 executables
add_executable(dynfilter_ros2 
  src/dynfilter_with_odom_ros2.cpp 
  src/DynObjFilter.cpp 
  src/DynObjCluster.cpp
)
ament_target_dependencies(dynfilter_ros2
  rclcpp
  geometry_msgs
  nav_msgs
  sensor_msgs
  std_msgs
  visualization_msgs
  tf2
  tf2_ros
  tf2_eigen
  pcl_conversions
  pcl_ros
)
target_link_libraries(dynfilter_ros2 
  ${PCL_LIBRARIES} 
  ${PYTHON_LIBRARIES} 
  ${OpenCV_LIBS} 
  TBB::tbb
)
target_include_directories(dynfilter_ros2 PRIVATE ${PYTHON_INCLUDE_DIRS})

add_executable(display_prediction_ros2 src/display_prediction_ros2.cpp)
ament_target_dependencies(display_prediction_ros2
  rclcpp
  geometry_msgs
  nav_msgs
  sensor_msgs
  std_msgs
  visualization_msgs
  tf2
  tf2_ros
  tf2_eigen
  pcl_conversions
  pcl_ros
)
target_link_libraries(display_prediction_ros2 
  ${PCL_LIBRARIES} 
  ${PYTHON_LIBRARIES} 
  ${OpenCV_LIBS} 
  TBB::tbb
)
target_include_directories(display_prediction_ros2 PRIVATE ${PYTHON_INCLUDE_DIRS})

add_executable(cal_recall_ros2 src/cal_recall_multi_ros2.cpp)
ament_target_dependencies(cal_recall_ros2
  rclcpp
  geometry_msgs
  sensor_msgs
  std_msgs
  visualization_msgs
  tf2
  tf2_ros
  pcl_conversions
  pcl_ros
)
target_link_libraries(cal_recall_ros2 
  ${PCL_LIBRARIES} 
  ${PYTHON_LIBRARIES} 
  ${OpenCV_LIBS} 
  TBB::tbb
)
target_include_directories(cal_recall_ros2 PRIVATE ${PYTHON_INCLUDE_DIRS})

# Install executables
install(TARGETS
  dynfilter_ros2
  display_prediction_ros2
  cal_recall_ros2
  DESTINATION lib/${PROJECT_NAME}
)

# Install directories
install(DIRECTORY
  launch
  config
  rviz
  DESTINATION share/${PROJECT_NAME}
)

if(BUILD_TESTING)
  find_package(ament_lint_auto REQUIRED)
  ament_lint_auto_find_test_dependencies()
endif()

ament_package()
